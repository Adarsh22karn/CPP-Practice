        description = desc;
        due_date = date;
        completed = comp;

    bool isCompleted() {
        return completed == 1;
    }
};

public:

    int numTasks;

    ToDoList() : numTasks(0) {}

    void addTask(string desc, string date, int comp) {
        if (numTasks < 10) {
            tasks[numTasks].setTask(desc, date, comp);
            numTasks++;
        }
    }

        for (int i = 0; i < numTasks; i++) {
            if (!tasks[i].isCompleted()) {
                cout << tasks[i].description << endl;
            }
        }
    }
};

        cin >> comp;

    todoList.displayIncompleteTasks();

    return 0;
}

#include <iostream>
#include <string>

using namespace std;

string generateAcronym(string phrase) {
    string acronym = "";
    bool newWord = true; // To track the start of a new word

    for (char c : phrase) {
        if (newWord && isalpha(c)) {
            acronym += toupper(c);
            newWord = false; // Set to false once we add the first character of a word
        } else if (!isalpha(c)) {
            newWord = true; // Set to true when we encounter non-alphabetic characters (word separators)
        }
    }

    return acronym;
}

int main() {
    string phrase;
    getline(cin, phrase);

    string acronym = generateAcronym(phrase);
    cout << acronym << endl;

    return 0;
}

Function Name: string encryptString

The first line of input consists of a string to be encrypted, str.

The second line of input consists of an integer, n representing the encryption key.
Output format :

The output displays the encrypted string as described in the problem statement.
Code constraints :

1 <= Length of the string, str <= 100

1 <= n <= 5
Sample test cases :
Input 1 :

welcome
2

Output 1 :

ygneqog

Input 2 :

Hello@123
4

LippsD567



#include <iostream>
#include <string>
using namespace std;

string encryptString(string str, int n) {
    for (char &ch : str) {
        if (isalpha(ch)) {
            if (islower(ch)) {
                // Encrypt lowercase letters
                ch = 'a' + (ch - 'a' + n) % 26;
            } else if (isupper(ch)) {
                // Encrypt uppercase letters
                ch = 'A' + (ch - 'A' + n) % 26;
            }
        }
    }
    return str;
}

int main() {
    string str;
    int n;

    // Input
    cin >> str;
    cin >> n;

    // Encrypt the string
    string encrypted_result = encryptString(str, n);

    // Output
    cout << encrypted_result << endl;

    return 0;
}

The third line of input consists of the replacing word.

If the word to be replaced is not present in the sentence, the output prints the original string as such.
Code constraints :

2 < Length of the sentence < 1000
Sample test cases :

Hi

Output 1 :

Hi world! 

Input 2 :

welcome
come

#include <iostream>
#include <string>

using namespace std;

string replaceWords(string sentence, string target_word, string replacement_word) {
    size_t found = sentence.find(target_word);
    
    // Check if the target word is present in the sentence
    if (found != string::npos) {
        // Replace all occurrences of the target word with the replacement word
        while (found != string::npos) {
            sentence.replace(found, target_word.length(), replacement_word);
            found = sentence.find(target_word, found + replacement_word.length());
        }
    }
    
    return sentence;
}

int main() {
    string sentence, target_word, replacement_word;
    
    // Input
    getline(cin, sentence);
    getline(cin, target_word);
    getline(cin, replacement_word);
    
    // Call the function and print the result
    string modified_sentence = replaceWords(sentence, target_word, replacement_word);
    cout << modified_sentence << endl;
    
    return 0;
}

#include <string>

class Employee {
public:
    string name;
    double rate_per_hour;
    int hours_worked;
    Employee() {} // Default constructor
    Employee(string n, double r, int h) {
        name = n;
        rate_per_hour = r;
        hours_worked = h;

        if (hours_worked <= 40) {
            return rate_per_hour * hours_worked;

        cout << "Employee Name: " << employees[i].name << endl;
        cout << "Salary: " << employees[i].calculateSalary() << endl;

Problem Statement


Raj wants to build a program that checks the strength of a given password. Implement a function checkPasswordStrength that takes a string representing a password as input and evaluates its strength based on criteria such as length, presence of uppercase letters, lowercase letters, numbers, and special characters. Use string objects to manipulate the password and perform the strength check.


-A password is classified as "Weak" if its length is less than 6 characters.

-A password is classified as "Medium" if its length is 6 or more characters, and it contains either an uppercase letter or a lowercase letter, along with at least one numeric digit.

-A password is classified as "Strong" if its length is 8 or more characters and it contains at least one uppercase letter, one lowercase letter, one numeric digit, and one special character.


Note: This kind of question will be helpful in clearing Capgemini tests.
Input format :

The input consists of a password that can be of any length and can contain any combination of characters.
Output format :

The program prints the strength of the password as "Weak," "Medium," or "Strong" based on the criteria mentioned in the question.
Code constraints :

The code assumes that the user will input a single line containing the password without any leading or trailing spaces.

1 <= Length of the password <= 25
Sample test cases :
Input 1 :

abcdef

Output 1 :

Weak

Input 2 :

Abcdef1

Output 2 :

Medium

Input 3 :

Abcdef1!

Output 3 :

Strong












#include <iostream>
#include <string>
#include <cctype>

using namespace std;

string checkPasswordStrength(const string& password) {
    int length = password.length();
    bool hasUppercase = false;
    bool hasLowercase = false;
    bool hasDigit = false;
    bool hasSpecialChar = false;

    for (int i = 0; i < length; i++) {
        char ch = password[i];

        if (isupper(ch)) {
            hasUppercase = true;
        } else if (islower(ch)) {
            hasLowercase = true;
        } else if (isdigit(ch)) {
            hasDigit = true;
        } else {
            hasSpecialChar = true;
        }
    }

    if (length < 6) {
        return "Weak";
    } else if (length >= 6 && (hasUppercase || hasLowercase) && hasDigit) {
        return "Medium";
    } else if (length >= 8 && hasUppercase && hasLowercase && hasDigit && hasSpecialChar) {
        return "Strong";
    } else {
        return "Weak";  // Default to Weak if none of the conditions are met
    }
}

int main() {
    string password;
    cin >> password;

    string strength = checkPasswordStrength(password);
    cout << strength << endl;

    return 0;
}
