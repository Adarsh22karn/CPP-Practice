Problem Statement


You have been given the task of building a program that converts a given sentence into LeetSpeak, a modified form of English using character substitutions.


Write a function toLeetSpeak that takes a sentence (string) as input and returns the LeetSpeak representation of the sentence, where only the vowels are replaced with their LeetSpeak counterparts.


Implement the function toLeetSpeak and write a main program to test it. The main program should prompt the user to enter a sentence and then display the corresponding LeetSpeak representation.


In this problem, 

-a/A is replaced with 4

-e/E is replaced with 3

-i/I is replaced with 1

-o/O is replaced with 0

-u?U s replaced with v


Note: This kind of question helps in clearing TCS tests.
Input format :

The input consists of a sentence.
Output format :

The program outputs the LeetSpeak representation of the sentence, where only the vowels are replaced with their LeetSpeak counterparts.
Code constraints :

The input sentence can contain alphabetic characters, digits, punctuation marks, and spaces.

The sentence can have a maximum length of 1,000 characters.
Sample test cases :
Input 1 :

Hello, world!

Output 1 :

H3ll0, w0rld!















#include <iostream>
#include <string>
using namespace std;

string toLeetSpeak(string sentence) {
    for (char &c : sentence) {
        switch (c) {
            case 'a': case 'A':
                c = '4';
                break;
            case 'e': case 'E':
                c = '3';
                break;
            case 'i': case 'I':
                c = '1';
                break;
            case 'o': case 'O':
                c = '0';
                break;
            case 'u': case 'U':
                c = 'v';
                break;
            default:
                break;
        }
    }
    return sentence;
}

int main() {
    string input_sentence;
    cout << "Enter a sentence: ";
    getline(cin, input_sentence);
    
    string leet_result = toLeetSpeak(input_sentence);
    cout << "LeetSpeak representation: " << leet_result << endl;
    
    return 0;
}

#include <iostream>

class OnlineShop {
    private:
        string shop_name;
        string* product_catalog;
        float* prices;
        int catalog_size;
        int num_products;

    public:
        OnlineShop(string name, int size) {
            shop_name = name;
            catalog_size = size;
            num_products = 0;
            product_catalog = new string[catalog_size];
            prices = new float[catalog_size];
        }

        ~OnlineShop() {
            delete[] product_catalog;
            delete[] prices;
        }

        void add_product(string product_name, float price) {
            if (num_products < catalog_size) {
                product_catalog[num_products] = product_name;
                prices[num_products] = price;
                num_products++;
                //cout << "Product added successfully." << endl;
            } else {
                cout << "Catalog is full. Product not added." << endl;
            }
        }

        void search_product(string product_name) {
            int index = -1;
            for (int i = 0; i < num_products; i++) {
                if (product_catalog[i] == product_name) {
                    index = i;
                    break;
                }
            }
            if (index != -1) {
                cout << "Product found: " << product_name << "; Price: " << prices[index] << endl;
            } else {
                cout << "Product not found." << endl;
            }
        }

        void display_products() {
            cout << "Available products:" << endl;
            for (int i = 0; i < num_products; i++) {
                cout << product_catalog[i] << " " << prices[i] << endl;
            }
        }

    int catalog_size;
    string product_name;

        shop.add_product(product_name, price);

Problem Statement


You have been tasked with creating a program that extracts sentences containing specific keywords from a given paragraph.


Write a function extractSentencesWithKeyword that takes a paragraph (string) and a keyword (string) as input and returns the sentences from the paragraph that contains the specified keyword.


Implement the function extractSentencesWithKeyword and write a main program to test it. The main program should prompt the user to enter a paragraph and a keyword, and then display the number of extracted sentences found and the extracted sentences that contain the keyword.


Note: This kind of question will be helpful in clearing Accenture tests.
Input format :

The first line consists of a paragraph.

The second line consists of the keyword which will be searched in the paragraph.
Output format :

The program outputs the number of extracted sentences found and the extracted sentences from the paragraph that contains the specified keyword in separate lines.

If no such keyword is present in the paragraph, the output prints 0.
Code constraints :

Length of the paragraph < 1000 characters
Sample test cases :
Input 1 :

The sun is hot. The ocean is vast. The sun shines bright.
sun

Output 1 :

2
The sun is hot.  The sun shines bright. 

Input 2 :

This is an example!
An

Output 2 :

0











#include <iostream>
#include <cstring>

using namespace std;

const int MAX_SENTENCES = 100;
const int MAX_LENGTH = 1000;

int main() {
    char paragraph[MAX_LENGTH];
    char keyword[MAX_LENGTH];
    char sentences[MAX_SENTENCES][MAX_LENGTH];
    int sentence_count = 0;

    // Get the paragraph and keyword from the user
    //cout << "Enter a paragraph: ";
    cin.getline(paragraph, MAX_LENGTH);
    //cout << "Enter a keyword: ";
    cin.getline(keyword, MAX_LENGTH);

    // Split the paragraph into sentences
    char *sentence = strtok(paragraph, ".?!");

    while (sentence != NULL && sentence_count < MAX_SENTENCES) {
        // Check if the sentence contains the keyword
        if (strstr(sentence, keyword) != NULL) {
            // Copy the sentence to the sentences array
            strcpy(sentences[sentence_count], sentence);
            sentence_count++;
        }

        // Get the next sentence
        sentence = strtok(NULL, ".?!");
    }

    // Print the extracted sentences
    if (sentence_count > 0) {
        cout << sentence_count << endl;
        for (int i = 0; i < sentence_count; i++) {
            cout <<  sentences[i]<<".";
        }
    } else {
        cout << "0" << endl;
    }

    return 0;
}
